# Path Guiding (Gap #32) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Cut the pre-denoise wall grain of daylight stills (12.8–16.4% on jobs 226/227/157) to ≤ 10% at the same 256 frames, with an unbiased, GPU-resident, deterministic path guide for diffuse bounces.

**Architecture:** A world-space hash grid (2^18 cells keyed by voxel + dominant normal axis), each cell an 8×8 equal-area (cos θ, φ) histogram of incident radiance in fixed-point `uint`, trained by the raygen from every path's own radiance estimate and read through a per-frame snapshot. At a diffuse-lobe draw the closest-hit mixes cosine sampling with histogram sampling (α = 0.5) and reports the mixture density through `combinedPdf`'s diffuse term, so every MIS consumer inherits it.

**Tech Stack:** GLSL (Vulkan RT stages), C++ (`render_raytracing.cpp`), doctest transcription tests, GDScript gates (`pt_validation.gd` harness), Python (`tools/daylight_validation.py`) for production acceptance.

func _check(p_name: String, p_value: float, p_expected: float, p_tol: float) -> void:
	_checks += 1
	var err := absf(p_value - p_expected) / maxf(absf(p_expected), 1e-9)
	print("  %s: %.6f vs %.6f (%+.2f%%, tol %.0f%%)" % [p_name, p_value, p_expected, 100.0 * (p_value / p_expected - 1.0), 100.0 * p_tol])
	if err > p_tol:
		_failures.append("%s: %.6f vs %.6f, %.2f%% off" % [p_name, p_value, p_expected, 100.0 * err])


func _check_true(p_name: String, p_ok: bool, p_detail: String = "") -> void:
	_checks += 1
	print("  %s %s %s" % ["PASS" if p_ok else "FAIL", p_name, p_detail])
	if not p_ok:
		_failures.append("%s %s" % [p_name, p_detail])

func _crop(p_img: Image, p_r: Array) -> float:
	var w := p_img.get_width()
	var h := p_img.get_height()
	var acc := 0.0
	var n := 0
	for y in range(int(h * float(p_r[0])), int(h * float(p_r[1]))):
		for x in range(int(w * float(p_r[2])), int(w * float(p_r[3]))):
			var c := p_img.get_pixel(x, y)
			acc += (c.r + c.g + c.b) / 3.0
			n += 1
	return acc / float(maxi(n, 1))


func _crop_channel(p_img: Image, p_r: Array, p_ch: int) -> float:
	var w := p_img.get_width()
	var h := p_img.get_height()
	var acc := 0.0
	var n := 0
	for y in range(int(h * float(p_r[0])), int(h * float(p_r[1]))):
		for x in range(int(w * float(p_r[2])), int(w * float(p_r[3]))):
			acc += p_img.get_pixel(x, y)[p_ch]
			n += 1
	return acc / float(maxi(n, 1))


func _rms(p_img: Image, p_ref: Image, p_r: Array) -> float:
	var w := p_img.get_width()
	var h := p_img.get_height()
	var acc := 0.0
	var n := 0
	for y in range(int(h * float(p_r[0])), int(h * float(p_r[1]))):
		for x in range(int(w * float(p_r[2])), int(w * float(p_r[3]))):
			var a := p_img.get_pixel(x, y)
			var b := p_ref.get_pixel(x, y)
			var d := (a.r + a.g + a.b) / 3.0 - (b.r + b.g + b.b) / 3.0
			acc += d * d
			n += 1
	return sqrt(acc / float(maxi(n, 1)))


## The daylight spec's grain statistic: RMS of the residual against a 5 x 5
## box mean, over the crop, divided by the crop mean.
func _grain(p_img: Image, p_r: Array) -> float:
	var w := p_img.get_width()
	var h := p_img.get_height()
	var y0 := int(h * float(p_r[0]))
	var y1 := int(h * float(p_r[1]))
	var x0 := int(w * float(p_r[2]))
	var x1 := int(w * float(p_r[3]))
	var acc := 0.0
	var mean_acc := 0.0
	var n := 0
	for y in range(y0 + 2, y1 - 2):
		for x in range(x0 + 2, x1 - 2):
			var box := 0.0
			for dy in range(-2, 3):
				for dx in range(-2, 3):
					var c := p_img.get_pixel(x + dx, y + dy)
					box += (c.r + c.g + c.b) / 3.0
			box /= 25.0
			var p := p_img.get_pixel(x, y)
			var v := (p.r + p.g + p.b) / 3.0
			acc += (v - box) * (v - box)
			mean_acc += v
			n += 1
	var mean := mean_acc / float(maxi(n, 1))
	return sqrt(acc / float(maxi(n, 1))) / maxf(mean, 1e-9)


func _max_abs_diff(p_a: Image, p_b: Image) -> float:
	var m := 0.0
	for y in range(p_a.get_height()):
		for x in range(p_a.get_width()):
			var a := p_a.get_pixel(x, y)
			var b := p_b.get_pixel(x, y)
			m = maxf(m, maxf(absf(a.r - b.r), maxf(absf(a.g - b.g), absf(a.b - b.b))))
	return m


func _save_pfm(p_img: Image, p_path: String) -> void:
	var w := p_img.get_width()
	var h := p_img.get_height()
	var f := FileAccess.open(p_path, FileAccess.WRITE)
	if f == null:
		return
	f.store_string("PF\n%d %d\n-1.0\n" % [w, h])
	var buf := StreamPeerBuffer.new()
	buf.big_endian = false
	for y in range(h - 1, -1, -1):
		for x in range(w):
			var c := p_img.get_pixel(x, y)
			buf.put_float(c.r)
			buf.put_float(c.g)
			buf.put_float(c.b)
	f.store_buffer(buf.data_array)
	f.close()


func _load_pfm(p_path: String) -> Image:
	var f := FileAccess.open(p_path, FileAccess.READ)
	if f == null:
		return null
	var magic := f.get_line()
	var dims := f.get_line().split(" ")
	f.get_line() # scale
	if magic != "PF" or dims.size() != 2:
		return null
	var w := int(dims[0])
	var h := int(dims[1])
	var img := Image.create(w, h, false, Image.FORMAT_RGBF)
	for y in range(h - 1, -1, -1):
		for x in range(w):
			var r := f.get_float()
			var g := f.get_float()
			var b := f.get_float()
			img.set_pixel(x, y, Color(r, g, b))
	return img

/**************************************************************************/
/* CPU mirror of raytracing_guiding_inc.glsl (Gap #32). The invariant:    */
/* the density used to SAMPLE a guided direction and the density REPORTED */
/* through combinedPdf must be one function. A drift is a bias that still */
/* looks like a plausible image; gate G0 catches it late, this catches it */
/* at build time.                                                          */
/**************************************************************************/

TEST_FORCE_LINK(test_pt_guiding)

#include "core/math/math_funcs.h"
#include "core/math/vector2.h"
#include "core/math/vector3.h"
#include <vector>

namespace TestPTGuiding {

static const int BINS = 64;
static const float BIN_SOLID_ANGLE = 4.0f * Math::PI / 64.0f;

// guide_bin_of_dir: 8 rows in cos(theta) over world +Y, 8 columns in phi = atan2(z, x).
static int bin_of_dir(const Vector3 &d) {
	float ct = CLAMP(d.y, -1.0f, 1.0f);
	int row = MIN(int((ct + 1.0f) * 0.5f * 8.0f), 7);
	float phi = Math::atan2(d.z, d.x);
	int col = MIN(int((phi + Math::PI) / (2.0f * Math::PI) * 8.0f), 7);
	return row * 8 + col;
}

// guide_dir_of_bin: uniform in the bin's (cos theta, phi) rectangle == uniform in solid angle.
static Vector3 dir_of_bin(int bin, const Vector2 &u) {
	int row = bin / 8;
	int col = bin % 8;
	float ct = -1.0f + (float(row) + u.x) * 0.25f;
	float phi = -Math::PI + (float(col) + u.y) * (2.0f * Math::PI / 8.0f);
	float st = Math::sqrt(MAX(1.0f - ct * ct, 0.0f));
	return Vector3(st * Math::cos(phi), ct, st * Math::sin(phi));
}

static Vector3 bin_center(int bin) {
	return dir_of_bin(bin, Vector2(0.5f, 0.5f));
}

// guide_nkey: dominant signed axis, 0..5.
static uint32_t nkey(const Vector3 &n) {
	Vector3 a = n.abs();
	int axis = (a.x >= a.y && a.x >= a.z) ? 0 : ((a.y >= a.z) ? 1 : 2);
	return uint32_t(axis) * 2u + (n[axis] < 0.0f ? 1u : 0u);
}

// guide_cell_of: lowbias32 over the voxel and the normal key, masked to 2^18.
static uint32_t cell_of(const Vector3 &p, const Vector3 &n, const Vector3 &origin, float cell) {
	int32_t vx = int32_t(Math::floor((p.x - origin.x) / cell));
	int32_t vy = int32_t(Math::floor((p.y - origin.y) / cell));
	int32_t vz = int32_t(Math::floor((p.z - origin.z) / cell));
	uint32_t h = uint32_t(vx) * 0x8DA6B343u ^ uint32_t(vy) * 0xD8163841u ^ uint32_t(vz) * 0xCB1AB31Fu ^ nkey(n) * 0x9E3779B9u;
	h ^= h >> 16;
	h *= 0x7FEB352Du;
	h ^= h >> 15;
	h *= 0x846CA68Bu;
	h ^= h >> 16;
	return h & (262144u - 1u);
}

// A cell's cosine-weighted bin weights for a lobe normal Nd, and their total.
static float bin_weight(const std::vector<uint32_t> &bins, int bin, const Vector3 &Nd) {
	return float(bins[bin]) * MAX(bin_center(bin).dot(Nd), 0.0f);
}

static float total_weight(const std::vector<uint32_t> &bins, const Vector3 &Nd) {
	float t = 0.0f;
	for (int b = 0; b < BINS; b++) {
		t += bin_weight(bins, b, Nd);
	}
	return t;
}

// guide_pdf: weight of L's bin over the total, per steradian.
static float guide_pdf(const std::vector<uint32_t> &bins, const Vector3 &Nd, float total, const Vector3 &L) {
	if (total <= 0.0f) {
		return 0.0f;
	}
	return bin_weight(bins, bin_of_dir(L), Nd) / (total * BIN_SOLID_ANGLE);
}

// guide_sample: CDF over the cosine-weighted bins with u.x, residual reused
// for the in-bin cos(theta) coordinate, u.y for phi.
static bool guide_sample(const std::vector<uint32_t> &bins, const Vector3 &Nd, float total, const Vector2 &u, Vector3 &r_L) {
	if (total <= 0.0f) {
		return false;
	}
	float target = u.x * total;
	float cdf = 0.0f;
	for (int b = 0; b < BINS; b++) {
		float w = bin_weight(bins, b, Nd);
		if (w <= 0.0f) {
			continue;
		}
		if (target < cdf + w || b == BINS - 1) {
			float residual = CLAMP((target - cdf) / w, 0.0f, 0.999999f);
			r_L = dir_of_bin(b, Vector2(residual, u.y));
			return true;
		}
		cdf += w;
	}
	return false;
}

// The diffuse lobe's density under guiding: (1 - a) cos/pi + a * guide.
static float mix_pdf(float alpha, float NdotLd, float guide) {
	return (1.0f - alpha) * MAX(NdotLd, 0.0f) / Math::PI + alpha * guide;
}

static Vector3 sphere_dir(float u, float v) {
	float ct = 1.0f - 2.0f * u;
	float st = Math::sqrt(MAX(1.0f - ct * ct, 0.0f));
	float phi = 2.0f * Math::PI * v;
	return Vector3(st * Math::cos(phi), ct, st * Math::sin(phi));
}

```glsl
// Path guiding for diffuse bounces (roadmap Gap #32, spec
// docs/superpowers/specs/2026-09-22-path-guiding-design.md).
//
// A world-space hash grid of directional histograms. Cell key = (voxel,
// dominant signed normal axis); bins = 8 x 8 in (cos theta, phi) about world
// +Y, exactly equal-area (4 pi / 64 sr each), so the pdf of a direction is
// its bin's weight over the cell's hemisphere total, per steradian.
//
// Two SSBOs: guide_train is written by the raygen (fixed-point atomics, so
// the sum is order-independent and the render repeatable); guide_read is a
// per-frame snapshot the closest-hit samples from, so the density that is
// sampled and the density that is reported through combinedPdf are one frozen
// table. Every function here is pure over those two buffers and the UBO.
//
// CPU twin: tests/servers/rendering/test_pt_guiding.cpp. Keep them literal.
#ifndef RT_GUIDING_INC
#define RT_GUIDING_INC

#define GUIDE_CELLS 262144u
#define GUIDE_BINS 64u
#define GUIDE_STRIDE 65u // 64 bins + the sample count
#define GUIDE_MIN_COUNT 32u
#define GUIDE_FIXED 1024.0
#define GUIDE_MAX_INV_PDF 64.0
#define GUIDE_RECORD 4
const float GUIDE_BIN_SOLID_ANGLE = 4.0 * PI / 64.0;

layout(set = 0, binding = 42, std430) buffer GuideTrainRW {
	uint guide_train[];
};
layout(set = 0, binding = 43, std430) readonly buffer GuideReadRO {
	uint guide_read[];
};

struct GuideState {
	bool active;
	float alpha;
	uint cell;
	float total; // cosine-weighted hemisphere total for Nd, raw fixed units
	vec3 Nd;
};

// Per-invocation vertex state. Inert unless shade_and_bounce looks a cell up;
// combinedPdf's diffuse term reads it so every MIS site sees one density.
GuideState g_guide = GuideState(false, 0.0, 0u, 0.0, vec3(0.0, 1.0, 0.0));

float guide_prob() {
	return get_rt_param(RT_PARAM_GUIDE_PROB);
}

bool guide_enabled() {
	return guide_prob() > 0.0 && get_rt_param(RT_PARAM_GUIDE_CELL) > 0.0;
}

uint guide_nkey(vec3 n) {
	vec3 a = abs(n);
	uint axis = (a.x >= a.y && a.x >= a.z) ? 0u : ((a.y >= a.z) ? 1u : 2u);
	return axis * 2u + ((n[axis] < 0.0) ? 1u : 0u);
}

uint guide_cell_of(vec3 p, vec3 n) {
	float cell = get_rt_param(RT_PARAM_GUIDE_CELL);
	vec3 o = vec3(get_rt_param(RT_PARAM_GUIDE_ORIGIN_X), get_rt_param(RT_PARAM_GUIDE_ORIGIN_X + 1), get_rt_param(RT_PARAM_GUIDE_ORIGIN_X + 2));
	ivec3 v = ivec3(floor((p - o) / cell));
	uint h = uint(v.x) * 0x8DA6B343u ^ uint(v.y) * 0xD8163841u ^ uint(v.z) * 0xCB1AB31Fu ^ guide_nkey(n) * 0x9E3779B9u;
	h ^= h >> 16;
	h *= 0x7FEB352Du;
	h ^= h >> 15;
	h *= 0x846CA68Bu;
	h ^= h >> 16;
	return h & (GUIDE_CELLS - 1u);
}

uint guide_bin_of_dir(vec3 d) {
	float ct = clamp(d.y, -1.0, 1.0);
	uint row = min(uint((ct + 1.0) * 0.5 * 8.0), 7u);
	float phi = atan(d.z, d.x);
	uint col = min(uint((phi + PI) / (2.0 * PI) * 8.0), 7u);
	return row * 8u + col;
}

vec3 guide_dir_of_bin(uint bin, vec2 u) {
	uint row = bin / 8u;
	uint col = bin % 8u;
	float ct = -1.0 + (float(row) + u.x) * 0.25;
	float phi = -PI + (float(col) + u.y) * (2.0 * PI / 8.0);
	float st = sqrt(max(1.0 - ct * ct, 0.0));
	return vec3(st * cos(phi), ct, st * sin(phi));
}

vec3 guide_bin_center(uint bin) {
	return guide_dir_of_bin(bin, vec2(0.5));
}

float guide_bin_weight(uint cell, uint bin, vec3 Nd) {
	return float(guide_read[cell * GUIDE_STRIDE + bin]) * max(dot(guide_bin_center(bin), Nd), 0.0);
}

GuideState guide_lookup(vec3 p, vec3 n, vec3 Nd) {
	GuideState g = GuideState(false, 0.0, 0u, 0.0, Nd);
	if (!guide_enabled()) {
		return g;
	}
	g.cell = guide_cell_of(p, n);
	if (guide_read[g.cell * GUIDE_STRIDE + GUIDE_BINS] < GUIDE_MIN_COUNT) {
		return g;
	}
	float total = 0.0;
	for (uint b = 0u; b < GUIDE_BINS; b++) {
		total += guide_bin_weight(g.cell, b, Nd);
	}
	if (total <= 0.0) {
		return g;
	}
	g.total = total;
	g.alpha = guide_prob();
	g.active = true;
	return g;
}

float guide_pdf(GuideState g, vec3 L) {
	if (!g.active) {
		return 0.0;
	}
	return guide_bin_weight(g.cell, guide_bin_of_dir(L), g.Nd) / (g.total * GUIDE_BIN_SOLID_ANGLE);
}

// combinedPdf's diffuse density. Identical to diffusePdf(NdotLd) when the
// vertex is not guided; the mixture when it is.
float guide_diffuse_pdf(float NdotLd, vec3 L) {
	float p_cos = NdotLd / PI;
	if (!g_guide.active) {
		return p_cos;
	}
	return (1.0 - g_guide.alpha) * p_cos + g_guide.alpha * guide_pdf(g_guide, L);
}

bool guide_sample(GuideState g, vec2 u, out vec3 L) {
	L = vec3(0.0);
	if (!g.active) {
		return false;
	}
	float target = u.x * g.total;
	float cdf = 0.0;
	for (uint b = 0u; b < GUIDE_BINS; b++) {
		float w = guide_bin_weight(g.cell, b, g.Nd);
		if (w <= 0.0) {
			continue;
		}
		if (target < cdf + w || b == GUIDE_BINS - 1u) {
			float residual = clamp((target - cdf) / w, 0.0, 0.999999);
			L = guide_dir_of_bin(b, vec2(residual, u.y));
			return true;
		}
		cdf += w;
	}
	return false;
}

// Training: value is luminance(L_in) / pdf, already clamped by the caller.
// Saturating: an add that would wrap is skipped (the cell has learnt what it
// needs); integer atomics keep the table order-independent.
void guide_train_vertex(vec3 p, vec3 n, vec3 dir, float value) {
	if (!(value > 0.0)) {
		return;
	}
	uint cell = guide_cell_of(p, n);
	uint idx = cell * GUIDE_STRIDE + guide_bin_of_dir(dir);
	uint add = uint(min(value * GUIDE_FIXED, 4.0e9));
	uint cur = guide_train[idx];
	if (cur < 0xFFFFFFFFu - add) {
		atomicAdd(guide_train[idx], add);
	}
	atomicAdd(guide_train[cell * GUIDE_STRIDE + GUIDE_BINS], 1u);
}

#endif // RT_GUIDING_INC

			guide_thr[guide_n] = ps.throughput; // after the BSDF weight and roulette
			guide_rad[guide_n] = ps.radiance; // through this vertex's own NEE
			guide_n++;
		}
```

After the loop (before `return ps.radiance;`), add:

```glsl
	if (guide_on) {
		const float clamp_ind = get_rt_param(RT_PARAM_CLAMP_INDIRECT);
		for (int k = 0; k < guide_n; k++) {
			vec3 thr = guide_thr[k];
			float thr_max = max(max(thr.r, thr.g), thr.b);
			if (thr_max <= 0.0) {
				continue; // the continuation carried no estimate
			}
			vec3 after = max(ps.radiance - guide_rad[k], vec3(0.0));
			vec3 L_in = after / max(thr, vec3(1e-6));
			float v = dot(L_in, vec3(0.2126, 0.7152, 0.0722));
			if (clamp_ind > 0.0) {
				v = min(v, clamp_ind);
			}
			float inv_pdf = min(1.0 / guide_pdf_k[k], GUIDE_MAX_INV_PDF);
			guide_train_vertex(guide_pos[k], guide_nrm[k], guide_dir[k], v * inv_pdf);
		}
	}
```

Note `ps.radiance - guide_rad[k]` is exact for the contributions added after vertex k because both are the same running sum; the `max(.., 0)` only guards float rounding.

The `volume_scatter_shade` `break` path also reaches this code: its radiance is "after" any recorded vertex, which is correct. A path that terminates INSIDE the closest-hit at vertex k (roulette, cap) has `guide_record` set with `mis_ray_pdf` possibly set too — that vertex is then recorded with `throughput_after`... check: on a roulette kill, shade_and_bounce sets `PATH_TERMINATED` and returns BEFORE `ps.mis_ray_pdf = combinedPdf(...)` is written (the pdf write is at the end). So `mis_ray_pdf` keeps its previous-vertex value or NONE. To be safe, in Step 4's guided/record code, set `ps.mis_ray_pdf = RT_MIS_PDF_NONE;` on every early `return` you add, and rely on the existing `is_path_terminated` `break` in raygen happening BEFORE the record (it does: the `break` is above `vec3 hit_pos = ...`). So a terminated vertex is never recorded. Good — document that in the comment.
